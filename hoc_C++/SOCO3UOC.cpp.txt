#include<bits/stdc++.h>
using namespace std;

const long long MAXN = 10000;
vector<bool> sangso(MAXN + 5, true);
vector<long long> dp(MAXN + 5, 0);

bool check(long long n){
    long long g = sqrtl(n);
    return sangso[g];
}

void sang(){
    sangso[0] = sangso[1] = false;
    for(int i=2;i*i<=MAXN;i++){
        if(sangso[i]){
            for(long long j=i*i;j<=MAXN;j+=i){
                sangso[j] = false;
            }
        }
    }

    for(long long i=0;i<=MAXN;i++){
        if(check(i)){
            dp[i] = dp[i-1] + 1;
        }else{
            dp[i] = dp[i-1];
        }
    }
}

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    long long a,b;
    cin >> a >> b;

    sang();

    cout << dp[b] - dp[a-1];
    return 0;
}
