#include<bits/stdc++.h>
using namespace std;

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    long long n, m;
    cin >> n >> m;

    map<long long, long long> he;

    long long dem = 0;
    for(int i=0;i<n;i++){
        long long x;
        cin >> x;
        he[x]++;
        if(he[x] > m){
            he[x] = -2e18;
        }
    }

    for(auto i : he){
        if(i.second < 0) continue;
        dem = max(i.second, dem);
    }
    
    cout << dem;
    return 0;
    // chx lam xg
}
